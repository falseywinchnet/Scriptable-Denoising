cmake_minimum_required(VERSION 3.20)
project(CleanupViewer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG a74efa0d5628b74adc0426af4c5710e287fa7c2c)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG f5befd2d29e66809cd1110a152e375a7f1981f06)
FetchContent_Declare(implot GIT_REPOSITORY https://github.com/epezent/implot.git GIT_TAG 18c72431f8265e2b0b5378a3a73d8a883b2175ff)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 65ee68451d8eb2b5f3a30b410476ab83deb3289b)
FetchContent_MakeAvailable(glfw imgui implot json)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
add_executable(CleanupViewer main.cpp
  ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp
  ${implot_SOURCE_DIR}/implot.cpp ${implot_SOURCE_DIR}/implot_items.cpp)
target_include_directories(CleanupViewer PRIVATE ${imgui_SOURCE_DIR} ${implot_SOURCE_DIR})
target_link_libraries(CleanupViewer PRIVATE glfw OpenGL::GL nlohmann_json::nlohmann_json Threads::Threads)
if(WIN32)
  target_link_libraries(CleanupViewer PRIVATE ws2_32)
  if(MINGW)
    target_link_options(CleanupViewer PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/VIEWER-LICENSES.txt" "CleanupViewer third-party notices\n\n")
foreach(license IN ITEMS "${glfw_SOURCE_DIR}/LICENSE.md" "${imgui_SOURCE_DIR}/LICENSE.txt" "${implot_SOURCE_DIR}/LICENSE" "${json_SOURCE_DIR}/LICENSE.MIT")
  file(READ "${license}" notice)
  get_filename_component(dependency_dir "${license}" DIRECTORY)
  get_filename_component(dependency "${dependency_dir}" NAME)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/VIEWER-LICENSES.txt" "${dependency}\n${notice}\n\n")
endforeach()
