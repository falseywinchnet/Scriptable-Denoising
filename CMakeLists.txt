cmake_minimum_required(VERSION 3.20)
project(CleanupExperiment VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python3 3.12 REQUIRED COMPONENTS Development)
find_package(Threads REQUIRED)
set(BFFT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(BFFT_BUILD_PROBES OFF CACHE BOOL "" FORCE)
set(BFFT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BFFT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BFFT_BUILD_HIGH_VISION OFF CACHE BOOL "" FORCE)
set(BFFT_ENABLE_AUTO_SIMD OFF CACHE BOOL "Portable Windows baseline" FORCE)
option(CLEANUP_USE_DIT_BFFT "Use validated bfft DIT forward/inverse paths" ON)
set(BFFT_USE_DIT_RFFT ${CLEANUP_USE_DIT_BFFT} CACHE BOOL "" FORCE)
set(BFFT_USE_DIT_RFFT_INVERSE ${CLEANUP_USE_DIT_BFFT} CACHE BOOL "" FORCE)
add_subdirectory(vendor/bfft EXCLUDE_FROM_ALL)
add_library(CleanupNative SHARED native/src/engine.cpp native/src/dip_transport.cpp native/src/recursive_smooth.cpp native/src/registered_guide.cpp research/true_superresolution/unrolled.cpp)
set(CLEANUP_GUIDE_THREADS "4" CACHE STRING "Maximum active guide lanes, including the audio caller (1..4)")
if(NOT CLEANUP_GUIDE_THREADS MATCHES "^[1-9][0-9]*$")
  message(FATAL_ERROR "CLEANUP_GUIDE_THREADS must be a positive integer")
endif()
if(CLEANUP_GUIDE_THREADS GREATER 4)
  message(STATUS "Capping the saved guide worker setting at the four-lane budget")
  set(CLEANUP_GUIDE_THREADS "4" CACHE STRING "Maximum active guide lanes, including the audio caller (1..4)" FORCE)
endif()
option(CLEANUP_DYNAMIC_GUIDE "Share chart and warp tasks across persistent guide workers" ON)
target_compile_definitions(CleanupNative PRIVATE CLEANUP_DYNAMIC_GUIDE=$<BOOL:${CLEANUP_DYNAMIC_GUIDE}>)
option(CLEANUP_PARALLEL_OBSERVATIONS "Prepare observation lattices using the existing guide worker pool" ON)
target_compile_definitions(CleanupNative PRIVATE CLEANUP_PARALLEL_OBSERVATIONS=$<BOOL:${CLEANUP_PARALLEL_OBSERVATIONS}>)
target_compile_definitions(CleanupNative PRIVATE CLEANUP_BUILD)
target_compile_definitions(CleanupNative PRIVATE CLEANUP_GUIDE_THREADS=${CLEANUP_GUIDE_THREADS})
option(CLEANUP_PROFILE_GUIDE "Time internal registration work (research only)" OFF)
target_compile_definitions(CleanupNative PRIVATE CLEANUP_PROFILE_GUIDE=$<BOOL:${CLEANUP_PROFILE_GUIDE}>)
target_include_directories(CleanupNative PUBLIC native/include)
target_include_directories(CleanupNative PRIVATE vendor/bfft/src)
option(CLEANUP_FUSED_RING_FORWARD "Exact fused Bruun/Hartley forward ring bank for 48-point charts" ON)
target_compile_definitions(CleanupNative PRIVATE CLEANUP_FUSED_RING_FORWARD=$<BOOL:${CLEANUP_FUSED_RING_FORWARD}>)
option(CLEANUP_PRUNED_RING_FORWARD "Experimental bounded peak queries without complete ring maps" OFF)
target_compile_definitions(CleanupNative PRIVATE CLEANUP_PRUNED_RING_FORWARD=$<BOOL:${CLEANUP_PRUNED_RING_FORWARD}>)
option(CLEANUP_EARLY_RING_FORWARD "Experimental rejection before the first Fourier axis completes" OFF)
target_compile_definitions(CleanupNative PRIVATE CLEANUP_EARLY_RING_FORWARD=$<BOOL:${CLEANUP_EARLY_RING_FORWARD}>)
set(CLEANUP_RING_LANES "4" CACHE STRING "Ring coefficients carried together (2, 4 or 8)")
if(NOT CLEANUP_RING_LANES MATCHES "^[248]$")
  message(FATAL_ERROR "CLEANUP_RING_LANES must be 2, 4 or 8")
endif()
target_compile_definitions(CleanupNative PRIVATE CLEANUP_RING_LANES=${CLEANUP_RING_LANES})
target_link_libraries(CleanupNative PRIVATE bfft_static Threads::Threads)
# These Fourier workspaces operate on bounded finite values. Avoid the compiler's
# general complex-overflow recovery and errno-setting sqrt slow paths; retain
# ordinary IEEE arithmetic/order (no fast-math or reassociation).
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  include(CheckCXXCompilerFlag)
  check_cxx_compiler_flag("-fcx-limited-range" CLEANUP_HAS_LIMITED_COMPLEX)
endif()
if(CLEANUP_HAS_LIMITED_COMPLEX)
  set_source_files_properties(native/src/registered_guide.cpp PROPERTIES
    COMPILE_OPTIONS "-fno-math-errno;-fcx-limited-range")
endif()
if(APPLE)
  target_link_libraries(CleanupNative PRIVATE Python3::Module)
  target_link_options(CleanupNative PRIVATE -undefined dynamic_lookup)
else()
  target_link_libraries(CleanupNative PRIVATE Python3::Python)
endif()
add_executable(cleanupctl native/src/control.cpp)
option(CLEANUP_BUILD_RESEARCH "Build isolated enhanced-magnitude oracle experiments" OFF)
if(CLEANUP_BUILD_RESEARCH)
  add_library(CleanupUnrolled SHARED research/true_superresolution/unrolled.cpp)
  target_link_libraries(CleanupUnrolled PRIVATE bfft_static)
endif()
if(WIN32)
  add_library(Cleanup SHARED native/src/loader.cpp)
  target_include_directories(Cleanup PUBLIC native/include)
  target_compile_definitions(Cleanup PRIVATE CLEANUP_BUILD)
  set_target_properties(Cleanup PROPERTIES PREFIX "")
  set_target_properties(CleanupNative PROPERTIES PREFIX "")
  target_link_libraries(CleanupNative PRIVATE ws2_32)
  target_link_libraries(cleanupctl PRIVATE ws2_32)
  if(MINGW)
    target_link_options(Cleanup PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(CleanupNative PRIVATE -static-libgcc -static-libstdc++)
    target_link_options(cleanupctl PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
